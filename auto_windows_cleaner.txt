@echo off
title Windows Safe Cleanup
color 0A

echo ==========================================
echo       WINDOWS SAFE CLEANUP
echo ==========================================
echo.
echo This will clean temporary/cache files.
echo Your personal files will NOT be deleted.
echo.
pause

:: Check for Administrator privileges
net session >nul 2>&1
if %errorlevel% neq 0 (
    echo.
    echo ERROR: Please run this file as Administrator.
    echo Right-click the .bat file ^> Run as administrator.
    pause
    exit /b
)

echo.
echo [1/7] Cleaning user TEMP files...
del /f /s /q "%TEMP%\*" >nul 2>&1
for /d %%D in ("%TEMP%\*") do rd /s /q "%%D" >nul 2>&1

echo [2/7] Cleaning Windows TEMP files...
del /f /s /q "%WINDIR%\Temp\*" >nul 2>&1
for /d %%D in ("%WINDIR%\Temp\*") do rd /s /q "%%D" >nul 2>&1

echo [3/7] Cleaning Prefetch cache...
del /f /s /q "%WINDIR%\Prefetch\*" >nul 2>&1

echo [4/7] Cleaning Windows thumbnail cache...
del /f /s /q "%LOCALAPPDATA%\Microsoft\Windows\Explorer\thumbcache_*.db" >nul 2>&1

echo [5/7] Cleaning Windows Update download cache...
net stop wuauserv >nul 2>&1
net stop bits >nul 2>&1

del /f /s /q "%WINDIR%\SoftwareDistribution\Download\*" >nul 2>&1
for /d %%D in ("%WINDIR%\SoftwareDistribution\Download\*") do rd /s /q "%%D" >nul 2>&1

net start bits >nul 2>&1
net start wuauserv >nul 2>&1

echo [6/7] Cleaning DirectX shader cache...
del /f /s /q "%LOCALAPPDATA%\D3DSCache\*" >nul 2>&1
for /d %%D in ("%LOCALAPPDATA%\D3DSCache\*") do rd /s /q "%%D" >nul 2>&1

echo [7/7] Emptying Recycle Bin...
powershell -NoProfile -Command "Clear-RecycleBin -Force -ErrorAction SilentlyContinue"

echo.
echo ==========================================
echo          CLEANUP COMPLETE
echo ==========================================
echo.
echo Temporary files and caches have been cleaned.
echo Some files may have been skipped because
echo they are currently being used by Windows.
echo.
pause
